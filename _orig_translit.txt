  const map = {
    '\u0905':'a','\u0906':'aa','\u0907':'i','\u0908':'ee','\u0909':'u','\u090a':'oo',
    '\u090b':'ri','\u090f':'e','\u0910':'ai','\u0913':'o','\u0914':'au',
    '\u0915':'k','\u0916':'kh','\u0917':'g','\u0918':'gh','\u0919':'ng',

    '\u093e':'aa','\u093f':'i','\u0940':'ee','\u0941':'u','\u0942':'oo',

}
